\section{Evidencias del prototipo}
\label{sec:anexo}

Este anexo documenta el prototipo construido en las semanas 1 a 6. El prototipo está programado e implementado en Python; aquí cada paso se presenta en pseudocódigo, es decir, describiendo en lenguaje sencillo lo que hace el programa, para que se entienda sin necesidad de leer el código. Junto a cada paso se muestra el archivo de Excel que produce, tal como lo vería la analista de planeamiento. El código fuente completo está en la carpeta \texttt{src/} del proyecto.

\subsection{Paso 1: limpieza}

\begin{algorithm}[H]
\caption{Limpieza del archivo}
\label{alg:limpiar}
\begin{algorithmic}[1]
\Require archivo exportado del sistema de compras
\Ensure líneas vigentes, líneas modificadas y resumen
\State verificar que existan las columnas obligatorias; si falta alguna, informar su nombre y detener
\State calcular la semana de cada línea a partir de su fecha de entrega
\State separar las líneas con indicador de borrado «L» (anuladas) de las vigentes
\ForAll{línea anulada}
    \State buscar una línea vigente de la misma tienda, producto y fecha de entrega
    \If{existe y su cantidad es distinta}
        \State registrarla como línea modificada (cantidad original y cantidad final)
    \EndIf
\EndFor
\State marcar como duplicadas las líneas vigentes con igual tienda, producto, fecha y cantidad
\State \Return líneas vigentes, líneas modificadas y resumen de conteos
\end{algorithmic}
\end{algorithm}

\subsection{Paso 2: señales}

\begin{algorithm}[H]
\caption{Cálculo de señales por tienda, producto y semana}
\label{alg:senales}
\begin{algorithmic}[1]
\Require líneas vigentes
\Ensure tabla de señales
\State sumar las cantidades por tienda, producto y semana para obtener $q$
\ForAll{tienda y producto}
    \ForAll{semana}
        \State $\hat q \gets$ mediana de $q$ en las 4 semanas anteriores (mínimo 2)
        \State $e \gets (q - \hat q) / \hat q$ \Comment{desviación respecto de lo habitual}
        \State $z \gets (q - \hat q)$ / variación habitual de la serie
        \State $\Delta \gets$ variación frente a la semana anterior
    \EndFor
\EndFor
\ForAll{semana}
    \State $a \gets$ mediana de $e$ de todas las series \Comment{movimiento común de la cadena}
    \State $u \gets e - a$ para cada serie \Comment{lo que se movió solo esa tienda}
\EndFor
\State contar cuántas presentaciones subieron más de 30\,\% en cada tienda y semana
\State excluir la última semana si el archivo termina antes del domingo
\State \Return tabla de señales
\end{algorithmic}
\end{algorithm}

\subsection{Paso 3: casos sospechosos}

\begin{algorithm}[H]
\caption{Selección de casos sospechosos y propuesta de etiqueta}
\label{alg:candidatos}
\begin{algorithmic}[1]
\Require tabla de señales
\Ensure hoja de revisión
\ForAll{tienda, producto y semana evaluable}
    \If{$u > 0{,}8$ \textbf{o} hay línea duplicada \textbf{o} $u < -0{,}6$}
        \State agregar a la lista de casos sospechosos con su motivo
        \If{subió sola (sin campaña y menos de 3 presentaciones en alza) \textbf{o} duplicada con alza mayor a 50\,\%}
            \State proponer «posible error»
        \Else
            \State proponer «normal»
        \EndIf
    \EndIf
\EndFor
\State dejar las columnas de etiqueta final y evidencia para la revisión manual
\State \Return hoja de revisión
\end{algorithmic}
\end{algorithm}

\subsection{Paso 4: reglas sencillas}

\begin{algorithm}[H]
\caption{Evaluación de las reglas sencillas}
\label{alg:reglas}
\begin{algorithmic}[1]
\Require tabla de señales con etiquetas propuestas
\Ensure métricas por regla y periodo
\State dividir las semanas en ajuste (hasta el 3 de agosto), validación (10 y 17 de agosto) y prueba (24 y 31 de agosto)
\ForAll{regla en \{semana anterior, habitual de la tienda, inusual sin ejemplos\}}
    \ForAll{periodo}
        \State contar alertas, casos sospechosos y aciertos
        \State precisión $\gets$ aciertos / alertas
        \State exhaustividad $\gets$ aciertos / casos sospechosos
        \State alertas por semana $\gets$ alertas / número de semanas
    \EndFor
\EndFor
\State \Return tabla de métricas
\end{algorithmic}
\end{algorithm}


\subsection{Versión por pedido}

\begin{algorithm}[H]
\caption{Señales de cada línea con la información disponible al registrarla}
\label{alg:linea}
\begin{algorithmic}[1]
\Require líneas vigentes ordenadas por fecha de registro
\Ensure señales por línea
\ForAll{línea $\ell$ registrada en la fecha $d$}
    \State considerar solo las líneas de la misma tienda y producto registradas hasta $d$
    \State $\hat q \gets$ mediana de los totales de las 4 semanas anteriores
    \State $p \gets$ proporción de la semana que la tienda suele llevar pedida hasta el día de entrega de $\ell$
    \State acumulado $\gets$ suma de la semana hasta el día de entrega de $\ell$
    \State $e \gets$ acumulado / $(\hat q \cdot p) - 1$
    \State tamaño $\gets$ cantidad de $\ell$ / mediana de las líneas anteriores
    \State duplicado $\gets$ existe una línea previa con igual fecha y cantidad
\EndFor
\State $a \gets$ mediana de $e$ de las líneas de la semana registradas hasta $d$
\State \Return señales por línea y días de anticipación respecto de la entrega
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Calibración de umbrales}
\label{alg:calibrar}
\begin{algorithmic}[1]
\Require señales por línea del periodo de ajuste
\Ensure umbrales de alerta
\ForAll{señal en \{acumulado de la semana, tamaño de la línea\}}
    \State ordenar los valores de mayor a menor
    \State elegir el umbral que deja como máximo 4 alertas por semana
\EndFor
\State aplicar los umbrales sin cambios al periodo de validación
\State \Return umbrales
\end{algorithmic}
\end{algorithm}


\subsection{Árbol de decisión y ventana del programa}

\begin{algorithm}[H]
\caption{Entrenamiento del árbol (botón «Entrenar modelo»)}
\label{alg:entrenar}
\begin{algorithmic}[1]
\Require Excel histórico de pedidos y Excel de etiquetas (hoja de revisión o lista de líneas con error)
\Ensure árbol guardado y sus reglas
\State calcular las señales de cada línea con el Algoritmo~\ref{alg:linea}
\If{el Excel tiene la columna \texttt{id\_linea}}
    \State etiqueta $\gets 1$ si el identificador orden-posición de la línea está en la lista; si no, $0$
\Else
    \State etiqueta $\gets 1$ si su caso está marcado como error, $0$ si es normal o no fue sospechoso
    \State excluir los casos marcados como duda
\EndIf
\State entrenar el árbol con ajuste y validación (profundidad 4, mínimo 5 líneas por hoja, peso 5 para la clase error); la prueba queda reservada
\State informar cuántos errores etiquetados se usaron
\State guardar el árbol
\State \Return reglas del árbol
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Revisión de pedidos (botón «Revisar pedidos»)}
\label{alg:revisar}
\begin{algorithmic}[1]
\Require Excel de pedidos de la semana y árbol guardado
\Ensure lista de líneas a revisar
\State calcular las señales de cada línea con el Algoritmo~\ref{alg:linea}
\ForAll{línea}
    \State recorrer el árbol desde la raíz respondiendo cada pregunta hasta llegar a una hoja
    \State $p \gets$ proporción de errores de esa hoja
    \If{$p \geq 0{,}5$}
        \State nivel $\gets$ «error probable» si $p \geq 0{,}8$; si no, «aviso»
        \State motivo $\gets$ las preguntas del recorrido, escritas como frase
    \EndIf
\EndFor
\State dejar una alerta por tienda, producto y semana (la de la primera línea marcada)
\State \Return alertas ordenadas por probabilidad
\end{algorithmic}
\end{algorithm}
